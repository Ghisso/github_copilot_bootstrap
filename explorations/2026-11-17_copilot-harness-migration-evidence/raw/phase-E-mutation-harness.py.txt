"""Targeted mutation checks for Phase E (scratch tool, evidence).

Usage: python3 -I mutation_checks.py

Copies the repository (without .git, .venv, and a top-level .claude) into a scratch folder, so the working copy is never
touched. For each mutation it changes ONE thing in the scratch copy of one shipped file, runs the tests that must
catch it, and restores the file. A mutation is KILLED when at least one of those tests fails (a timeout counts: the
process group is killed). A mutation that no test catches is SURVIVED. A control run of the unmutated scratch copy
must pass first.
"""

class Mutation:
    name: str
    file: str
    old: str
    new: str
    tests: list[str]
    expect: str  # "killed" or "survives"
    why: str = ""

def run_tests(args: list[str], timeout: int = 240) -> tuple[bool, list[str], float]:
    """Run pytest in the scratch copy. Returns (all passed, failing ids, seconds)."""
    start = time.monotonic()
    process = subprocess.Popen(
        [str(PYTHON), "-m", "pytest", "-q", "--tb=no", "-rf", "-p", "no:cacheprovider", *args],
        cwd=SCRATCH, env=CLEAN_ENV, text=True, stdout=subprocess.PIPE, stderr=subprocess.STDOUT, start_new_session=True,
    )
    try:
        out, _ = process.communicate(timeout=timeout)
        timed_out = False
    except subprocess.TimeoutExpired:
        os.killpg(process.pid, signal.SIGKILL)
        out, _ = process.communicate()
        timed_out = True
    try:
        os.killpg(process.pid, signal.SIGKILL)
    except ProcessLookupError:
        pass
    failing = re.findall(r"^FAILED (\S+)", out, flags=re.M)
    if timed_out:
        failing.append(f"(no result after {timeout} s: the run hung and was killed)")
    return (process.returncode == 0 and not timed_out), failing, time.monotonic() - start


def main() -> int:
    if SCRATCH.exists():
        shutil.rmtree(SCRATCH)
    SCRATCH.mkdir(parents=True)
    skip = {".git", ".venv", ".claude", ".mypy_cache", ".ruff_cache", ".pytest_cache", "node_modules"}
    for entry in REPO.iterdir():
        if entry.name in skip:
            continue
        target = SCRATCH / entry.name
        if entry.is_dir():
            shutil.copytree(entry, target, symlinks=True, ignore=shutil.ignore_patterns("__pycache__"))
        else:
            shutil.copy2(entry, target)
    print("Targeted mutation checks, scratch copy:", SCRATCH)
    print("Working copy untouched; one thing changed per mutation in the scratch copy of one shipped file.")
    print()
    bad = 0
    for label, args in (("tests/test_hook_gates.py", GATES), ("tests/test_validate_targets.py", VALIDATE), ("tests/test_check_copilot_runtime.py", PROBE_TESTS)):
        passed, failing, seconds = run_tests(args)
        print(f"CONTROL (no mutation) {label} {' '.join(args[1:])!r}: {'all passed' if passed else 'FAILED ' + str(failing)} ({seconds:.1f} s)")
        bad += not passed
    print()
    print("| # | mutation | file | result | tests that failed (first 3) |")
    print("|---|---|---|---|---|")
    for number, mutation in enumerate(MUTATIONS, 1):
        path = SCRATCH / mutation.file
        original = path.read_text(encoding="utf-8")
        assert original.count(mutation.old) == 1, f"{mutation.name}: {mutation.old!r} occurs {original.count(mutation.old)} times"
        try:
            path.write_text(original.replace(mutation.old, mutation.new), encoding="utf-8")
            passed, failing, seconds = run_tests(mutation.tests)
        finally:
            path.write_text(original, encoding="utf-8")
        killed = not passed
        result = "KILLED" if killed else "SURVIVED"
        verdict_ok = killed == (mutation.expect == "killed")
        bad += not verdict_ok
        shown = "; ".join(failing[:3]) + (f" (+{len(failing) - 3} more)" if len(failing) > 3 else "")
        note = "" if verdict_ok else " <-- UNEXPECTED"
        print(f"| {number} | {mutation.name} | {mutation.file} | {result}{note} | {shown or '(none)'} |")
        if mutation.why:
            print(f"|   | note: {mutation.why} | | | |")
    print()
    print(f"Unexpected outcomes: {bad}")
    return 1 if bad else 0


sys.exit(main())
